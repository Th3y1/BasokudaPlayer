// --- KONFIGURACJA --- const ADMIN_EMAIL = "dogdeyy@gmail.com"; // --- ELEMENTY DOM --- const loginSection = document.getElementById("login-section"); // Twój kontener z przyciskami logowania const appContent = document.getElementById("app-content"); // Twój kontener z główną treścią strony const userStatus = document.getElementById("user-status"); // Napis np. "Zalogowano: Discord (nick)" const devBtn = document.getElementById("dev-tools-btn"); // Przycisk "Narzędzia Dev" // --- 1. OBSŁUGA POWROTU Z DISCORDA (OAuth2) --- function handleDiscordCallback() { // Discord po zalogowaniu przysyła token w adresie URL po znaku '#' const hash = window.location.hash.substring(1); const params = new URLSearchParams(hash); const accessToken = params.get("access_token"); if (accessToken) { // Czyścimy hash z adresu URL, żeby link wyglądał czysto window.history.replaceState({}, document.title, window.location.pathname); // Pobieramy dane użytkownika z API Discorda fetch("https://discord.com/api/users/@me", { headers: { Authorization: `Bearer ${accessToken}` } }) .then(res => res.json()) .then(userData => { if (userData.id) { // Zapisujemy sesję w pamięci przeglądarki const sessionUser = { name: userData.global_name || userData.username, email: userData.email || `${userData.username}@discord.user`, // Jeśli zakres (scope) nie zawierał maila provider: "Discord", avatar: userData.avatar ? `https://cdn.discordapp.com/avatars/${userData.id}/${userData.avatar}.png` : null }; localStorage.setItem("loggedUser", JSON.stringify(sessionUser)); initApp(sessionUser); } }) .catch(err => console.error("Błąd podczas pobierania profilu Discord:", err)); } } // --- 2. INICJALIZACJA WIDOKU STRONY --- function initApp(user) { if (!user) { // Nikt nie jest zalogowany -> pokaż formularz logowania if (loginSection) loginSection.style.display = "block"; if (appContent) appContent.style.display = "none"; return; } // Użytkownik jest zalogowany -> pokaż treść strony if (loginSection) loginSection.style.display = "none"; if (appContent) appContent.style.display = "block"; // Wyświetl status zalogowania if (userStatus) { userStatus.textContent = `Zalogowano: ${user.provider} (${user.email})`; } // ZABEZPIECZENIE DEV TOOLS: // Przycisk Dev pokaże się TYLKO dla Twojego maila if (devBtn) { if (user.email.toLowerCase() === ADMIN_EMAIL.toLowerCase()) { devBtn.style.display = "inline-block"; } else { devBtn.style.display = "none"; } } } // --- 3. WYLOGOWANIE --- function logout() { localStorage.removeItem("loggedUser"); window.location.reload(); } // --- 4. START SKRYPTU --- document.addEventListener("DOMContentLoaded", () => { // A. Sprawdź, czy właśnie wróciliśmy z autoryzacji Discorda if (window.location.hash.includes("access_token")) { handleDiscordCallback(); return; } // B. Sprawdź, czy użytkownik jest już zapisany w pamięci (Local Storage) const savedUser = localStorage.getItem("loggedUser"); if (savedUser) { initApp(JSON.parse(savedUser)); } else { initApp(null); } });
